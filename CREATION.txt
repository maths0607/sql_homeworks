CREATE DATABASE dostavka;

CREATE TABLE client(
id SERIAL PRIMARY KEY,
name VARCHAR(52) NOT NULL,
vip BOOLEAN DEFAULT FALSE,
address VARCHAR(52) NOT NULL,
contact VARCHAR(100) NOT NULL
);

CREATE TABLE deliverer(
id SERIAL PRIMARY KEY,
name VARCHAR(52) NOT NULL,
rate SMALLINT DEFAULT NULL,
vehicule VARCHAR(52),
last_position VARCHAR(52) NOT NULL,
contact VARCHAR(100) NOT NULL
);

CREATE TABLE resto(
id SERIAL PRIMARY KEY,
name VARCHAR(52) NOT NULL,
rate SMALLINT DEFAULT NULL,
address VARCHAR(52) NOT NULL,
contact VARCHAR(100) NOT NULL,
open BOOLEAN DEFAULT TRUE
);

CREATE TABLE orders(
id SERIAL PRIMARY KEY,
amount DECIMAL(10,2) DEFAULT NULL,
id_client INT NOT NULL,
id_deliverer INT,
id_resto INT,
ordering_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
delivered_time TIMESTAMP DEFAULT NULL,
rate SMALLINT DEFAULT NULL
);

CREATE TABLE dish(
id SERIAL PRIMARY KEY,
name VARCHAR(52) NOT NULL,
description VARCHAR(100) NOT NULL,
price DECIMAL(10,2) NOT NULL,
rate SMALLINT DEFAULT NULL
);

CREATE TABLE order_dish(
id_order INT NOT NULL,
id_dish INT NOT NULL,
quantity_dish INT NOT NULL DEFAULT 1,
PRIMARY KEY(id_order,id_dish)
);

CREATE TABLE resto_dish(
id_resto INT NOT NULL,
id_dish INT NOT NULL,
has BOOLEAN,
PRIMARY KEY(id_resto,id_dish)
);
